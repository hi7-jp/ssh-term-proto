+/**
+ * Character-grid terminal view: draws a [TerminalEmulator]'s buffer with a
+ * monospace font and turns key / IME input into bytes for the SSH session.
+ *
+ * ## Input design (chosen: dedicated key-input View, no EditText)
+ * The old "one-line EditText + Send button" is gone. This view itself is the
+ * text editor as far as the IME is concerned ([onCheckIsTextEditor] +
+ * [onCreateInputConnection]):
+ *  - Soft keyboard characters arrive via InputConnection.commitText and are
+ *    forwarded immediately, one commit at a time (no Send button, no line

+ *  - Soft keyboard special keys (Enter, Backspace) and every hardware key

+ *  - Keys a phone keyboard lacks (Esc, Tab, Ctrl, arrows ...) are provided by
+ *    the extra-key row in MainActivity, which calls [sendText] / [sendKey].
+ *
+ * Integration with the existing UI: [MainActivity] puts this view where the
+ * ScrollView+TextView log was, wires [onInput] to the SSH output stream and

+        val newCols = max(2, (usableW / cellW).toInt())
+        val newRows = max(1, (usableH / cellH).toInt())

+            emulator.resize(newCols, newRows)
+            scrollOffset = min(scrollOffset, emulator.buffer.scrollbackSize)
+            onResize?.invoke(newCols, newRows, w, h)
+        }
+    }

+    override fun onDraw(canvas: Canvas) {
+        val buf = emulator.buffer
+        val defBg = TerminalColors.DEFAULT_BG
+        canvas.drawColor(defBg)
+
+        val left = paddingLeft.toFloat()
+        val top = paddingTop.toFloat()
+        val focused = hasFocus()
+        val showCursor = emulator.cursorVisible && scrollOffset == 0
+        var lastStyle = -1
+        textPaint.typeface = normalTypeface
+        textPaint.textSkewX = 0f
+
+        for (r in 0 until buf.rows) {
+            val row = buf.getRow(r - scrollOffset) ?: continue
+            val y = top + r * cellH
+            val n = min(buf.cols, row.size)
+            for (c in 0 until n) {
+                val attrs = row.attrs[c]
+                if ((attrs and TerminalAttr.WIDE_TAIL) != 0) continue // painted with its head
+
+                val wide = c + 1 < n && (row.attrs[c + 1] and TerminalAttr.WIDE_TAIL) != 0
+                val cw = if (wide) cellW * 2 else cellW
+                val x = left + c * cellW
+
+                var fgC = TerminalColors.resolve(row.fg[c], TerminalColors.DEFAULT_FG)
+                var bgC = TerminalColors.resolve(row.bg[c], defBg)
+                if ((attrs and TerminalAttr.INVERSE) != 0) {
+                    val t = fgC
+                    fgC = bgC
+                    bgC = t
+                }
+
+                val isCursor = showCursor && r == buf.cursorY && c == buf.cursorX
+                if (isCursor && focused && blinkOn) {
+                    // Block cursor = colours swapped relative to what is displayed.
+                    val t = fgC
+                    fgC = bgC
+                    bgC = t
+                }
+
+                if (bgC != defBg) {
+                    fillPaint.color = bgC
+                    canvas.drawRect(x, y, x + cw, y + cellH, fillPaint)
+                }
+
+                val cp = row.chars[c]
+                if (cp != 0x20 && (attrs and TerminalAttr.INVISIBLE) == 0) {
+                    if ((attrs and TerminalAttr.DIM) != 0) {
+                        fgC = (fgC and 0x00FFFFFF) or (0x99 shl 24)
+                    }
+                    val style = attrs and (TerminalAttr.BOLD or TerminalAttr.ITALIC)
+                    if (style != lastStyle) {
+                        textPaint.typeface =
+                            if ((style and TerminalAttr.BOLD) != 0) boldTypeface else normalTypeface
+                        textPaint.textSkewX = if ((style and TerminalAttr.ITALIC) != 0) -0.25f else 0f
+                        lastStyle = style
+                    }
+                    textPaint.color = fgC
+                    val len = Character.toChars(cp, charBuf, 0)
+                    canvas.drawText(charBuf, 0, len, x, y + baseline, textPaint)
+                }
+
+                if ((attrs and TerminalAttr.UNDERLINE) != 0) {
+                    fillPaint.color = fgC
+                    canvas.drawRect(x, y + cellH - lineThickness, x + cw, y + cellH, fillPaint)
+                }
+                if ((attrs and TerminalAttr.STRIKE) != 0) {
+                    fillPaint.color = fgC
+                    val sy = y + cellH / 2
+                    canvas.drawRect(x, sy, x + cw, sy + lineThickness, fillPaint)
+                }
+
+                if (isCursor && !focused) {
+                    // Hollow box when the view doesn't have input focus.
+                    strokePaint.color = TerminalColors.DEFAULT_FG
+                    canvas.drawRect(x + 1f, y + 1f, x + cw - 1f, y + cellH - 1f, strokePaint)
+                }
+            }
+        }
+
+        drawComposing(canvas, left, top)
+    }

+    /** Draws IME pre-edit text (e.g. unconverted kana) at the cursor, underlined. */
+    private fun drawComposing(canvas: Canvas, left: Float, top: Float) {
+        if (composing.isEmpty() || scrollOffset != 0) return
+        val buf = emulator.buffer
+        var cx = buf.cursorX
+        val y = top + buf.cursorY * cellH
+        textPaint.typeface = normalTypeface
+        textPaint.textSkewX = 0f
+        var idx = 0
+        while (idx < composing.length) {
+            val cp = composing.codePointAt(idx)
+            idx += Character.charCount(cp)
+            val w = CharWidth.of(cp)
+            if (w == 0) continue
+            if (cx + w > buf.cols) break
+            val x = left + cx * cellW
+            fillPaint.color = 0xFF3A3A3A.toInt()
+            canvas.drawRect(x, y, x + w * cellW, y + cellH, fillPaint)
+            textPaint.color = TerminalColors.DEFAULT_FG
+            val len = Character.toChars(cp, charBuf, 0)
+            canvas.drawText(charBuf, 0, len, x, y + baseline, textPaint)
+            fillPaint.color = TerminalColors.DEFAULT_FG
+            canvas.drawRect(x, y + cellH - lineThickness, x + w * cellW, y + cellH, fillPaint)
+            cx += w
+        }
+    }

+    /** Returns true when the key was turned into terminal input. */
+    private fun handleKey(keyCode: Int, event: KeyEvent): Boolean {
+        val shift = event.isShiftPressed
+        val alt = event.isAltPressed
+        val ctrl = event.isCtrlPressed
+
+        val special = keySequence(keyCode, shift, alt, ctrl)
+        if (special != null) {
+            sendText(special)
+            return true
+        }
+
+        // Printable key: ask the key map for the character *without* Ctrl/Alt
+        // applied, then apply them ourselves the way a terminal expects.
+        val meta = event.metaState and (KeyEvent.META_CTRL_MASK or KeyEvent.META_ALT_MASK).inv()
+        val uc = event.getUnicodeChar(meta)
+        if (uc <= 0) return false // modifier keys, Back, volume, dead keys ...
+
+        var s = String(Character.toChars(uc))
+        if (ctrl && uc < 0x10000) {
+            val code = ctrlCode(uc.toChar())
+            if (code >= 0) s = code.toChar().toString()
+        }
+        if (alt) s = "\u001b$s" // Alt = Meta = ESC prefix
+        sendText(s)
+        return true
+    }
+
+    /** xterm-style sequences for non-printing keys, or null if not one of them. */
+    private fun keySequence(keyCode: Int, shift: Boolean, alt: Boolean, ctrl: Boolean): String? {
+        val mods = 1 + (if (shift) 1 else 0) + (if (alt) 2 else 0) + (if (ctrl) 4 else 0)
+        val app = emulator.appCursorKeys
+        return when (keyCode) {
+            KeyEvent.KEYCODE_ENTER, KeyEvent.KEYCODE_NUMPAD_ENTER ->
+                if (alt) "\u001b\r" else "\r" // Alt+Enter = ESC CR (newline in many TUIs)
+            KeyEvent.KEYCODE_DEL -> if (alt) "\u001b\u007f" else "\u007f"
+            KeyEvent.KEYCODE_FORWARD_DEL -> tildeSeq(3, mods)
+            KeyEvent.KEYCODE_TAB -> if (shift) "\u001b[Z" else "\t"
+            KeyEvent.KEYCODE_ESCAPE -> "\u001b"
+            KeyEvent.KEYCODE_DPAD_UP -> cursorSeq('A', mods, app)
+            KeyEvent.KEYCODE_DPAD_DOWN -> cursorSeq('B', mods, app)
+            KeyEvent.KEYCODE_DPAD_RIGHT -> cursorSeq('C', mods, app)
+            KeyEvent.KEYCODE_DPAD_LEFT -> cursorSeq('D', mods, app)
+            KeyEvent.KEYCODE_MOVE_HOME -> cursorSeq('H', mods, app)
+            KeyEvent.KEYCODE_MOVE_END -> cursorSeq('F', mods, app)
+            KeyEvent.KEYCODE_PAGE_UP -> tildeSeq(5, mods)
+            KeyEvent.KEYCODE_PAGE_DOWN -> tildeSeq(6, mods)
+            KeyEvent.KEYCODE_INSERT -> tildeSeq(2, mods)
+            KeyEvent.KEYCODE_F1 -> ss3Seq('P', mods)
+            KeyEvent.KEYCODE_F2 -> ss3Seq('Q', mods)
+            KeyEvent.KEYCODE_F3 -> ss3Seq('R', mods)
+            KeyEvent.KEYCODE_F4 -> ss3Seq('S', mods)
+            KeyEvent.KEYCODE_F5 -> tildeSeq(15, mods)
+            KeyEvent.KEYCODE_F6 -> tildeSeq(17, mods)
+            KeyEvent.KEYCODE_F7 -> tildeSeq(18, mods)
+            KeyEvent.KEYCODE_F8 -> tildeSeq(19, mods)
+            KeyEvent.KEYCODE_F9 -> tildeSeq(20, mods)
+            KeyEvent.KEYCODE_F10 -> tildeSeq(21, mods)
+            KeyEvent.KEYCODE_F11 -> tildeSeq(23, mods)
+            KeyEvent.KEYCODE_F12 -> tildeSeq(24, mods)
+            else -> null
+        }
+    }
+
+    private fun cursorSeq(ch: Char, mods: Int, app: Boolean): String = when {
+        mods > 1 -> "\u001b[1;$mods$ch"
+        app -> "\u001bO$ch"
+        else -> "\u001b[$ch"
+    }
+
+    private fun tildeSeq(n: Int, mods: Int): String =
+        if (mods > 1) "\u001b[$n;$mods~" else "\u001b[$n~"
+
+    private fun ss3Seq(ch: Char, mods: Int): String =
+        if (mods > 1) "\u001b[1;$mods$ch" else "\u001bO$ch"
+
+    /** Control-code for Ctrl+[c], or -1 when Ctrl has no meaning for it. */
+    private fun ctrlCode(c: Char): Int = when (c) {
+        in 'a'..'z' -> c - 'a' + 1
+        in 'A'..'Z' -> c - 'A' + 1
+        '@', ' ' -> 0
+        '[' -> 27
+        '\\' -> 28
+        ']' -> 29
+        '^' -> 30
+        '_', '/' -> 31
+        '?' -> 127
+        else -> -1
+    }
+
+    private fun emit(bytes: ByteArray) {
+        if (scrollOffset != 0) {
+            scrollOffset = 0 // typing jumps back to the live screen
+        }
+        resetBlink()
+        invalidate()
+        onInput?.invoke(bytes)
+    }
+
+    /**
+     * Bridges the soft keyboard. Only the methods an IME really uses are
+     * implemented; there is no backing Editable because the "text" lives on the
+     * remote side.
+     */
+    private inner class TerminalInputConnection(view: View) : BaseInputConnection(view, false) {

+        override fun commitText(text: CharSequence?, newCursorPosition: Int): Boolean {

+            return true
+        }

+        override fun setComposingText(text: CharSequence?, newCursorPosition: Int): Boolean {

+            return true
+        }
+
+        override fun finishComposingText(): Boolean {
+            if (composing.isNotEmpty()) {
+                val t = composing
+                composing = ""
+                sendText(t)
+                invalidate()
+            }
+            return true
+        }
+
+        override fun deleteSurroundingText(beforeLength: Int, afterLength: Int): Boolean {
+            if (composing.isNotEmpty()) {
+                composing = composing.dropLast(beforeLength)
+                invalidate()
+            } else {
+                repeat(beforeLength) { sendText("\u007f") }
+            }
+            return true
+        }
+
+        override fun deleteSurroundingTextInCodePoints(beforeLength: Int, afterLength: Int): Boolean =
+            deleteSurroundingText(beforeLength, afterLength)
+
+        override fun sendKeyEvent(event: KeyEvent): Boolean {
+            if (event.action == KeyEvent.ACTION_DOWN) {
+                handleKey(event.keyCode, event)
+            }
+            return true
+        }
+
+        override fun performEditorAction(actionCode: Int): Boolean {
+            sendText("\r")
+            return true
+        }

+        override fun getTextAfterCursor(n: Int, flags: Int): CharSequence? = ""
+        override fun getSelectedText(flags: Int): CharSequence? = null
+        override fun setComposingRegion(start: Int, end: Int): Boolean = true
+        override fun setSelection(start: Int, end: Int): Boolean = true
+        override fun beginBatchEdit(): Boolean = true
+        override fun endBatchEdit(): Boolean = true
+    }

+/**
+ * Minimal wcwidth: 0 for combining / zero-width code points, 2 for East Asian
+ * wide + emoji, 1 otherwise. Ambiguous-width characters (box drawing, ●, ○ ...)
+ * are treated as 1 cell, which is what most Western-locale TUIs assume.
+ *
+ * Without this, Japanese output would desync the cursor column and break the
+ * box borders of full-screen TUIs.
+ */
+object CharWidth {
+
+    fun of(cp: Int): Int {
+        if (cp < 0x300) return 1
+        if (isZeroWidth(cp)) return 0
+        return if (isWide(cp)) 2 else 1
+    }
+
+    private fun isZeroWidth(cp: Int): Boolean =
+        (cp in 0x0300..0x036F) ||
+            (cp in 0x1AB0..0x1AFF) ||
+            (cp in 0x1DC0..0x1DFF) ||
+            (cp in 0x200B..0x200F) ||
+            (cp in 0x202A..0x202E) ||
+            cp == 0x2060 ||
+            (cp in 0x20D0..0x20FF) ||
+            (cp in 0xFE00..0xFE0F) ||
+            cp == 0xFEFF ||
+            (cp in 0xE0100..0xE01EF)
+
+    private fun isWide(cp: Int): Boolean =
+        (cp in 0x1100..0x115F) ||     // Hangul Jamo
+            (cp in 0x2E80..0x303E) || // CJK radicals, punctuation (incl. U+3000)
+            (cp in 0x3041..0x33FF) || // Hiragana, Katakana, CJK symbols
+            (cp in 0x3400..0x4DBF) || // CJK Ext A
+            (cp in 0x4E00..0x9FFF) || // CJK Unified
+            (cp in 0xA000..0xA4CF) || // Yi
+            (cp in 0xAC00..0xD7A3) || // Hangul syllables
+            (cp in 0xF900..0xFAFF) || // CJK compatibility ideographs
+            (cp in 0xFE30..0xFE6F) || // CJK compatibility forms
+            (cp in 0xFF00..0xFF60) || // Fullwidth forms
+            (cp in 0xFFE0..0xFFE6) ||
+            (cp in 0x1F300..0x1F64F) || // emoji
+            (cp in 0x1F680..0x1F6FF) ||
+            (cp in 0x1F900..0x1F9FF) ||
+            (cp in 0x1FA70..0x1FAFF) ||
+            (cp in 0x20000..0x3FFFD)    // CJK Ext B..

+/** Per-cell attribute bits stored in [TerminalRow.attrs]. */
+object TerminalAttr {
+    const val BOLD = 1
+    const val DIM = 2
+    const val ITALIC = 4
+    const val UNDERLINE = 8
+    const val INVERSE = 16
+    const val STRIKE = 32
+    const val INVISIBLE = 64
+
+    /** Marks the right half of a double-width character (never drawn itself). */
+    const val WIDE_TAIL = 128
+}

+/**
+ * One row of cells. Parallel IntArrays instead of a Cell object per column,
+ * so scrolling a 80x40 screen allocates nothing.
+ */
+class TerminalRow(cols: Int) {
+    var chars: IntArray = IntArray(cols) { SPACE }   // code points
+    var fg: IntArray = IntArray(cols)                // see TerminalColors
+    var bg: IntArray = IntArray(cols)
+    var attrs: IntArray = IntArray(cols)             // TerminalAttr bits
+
+    val size: Int get() = chars.size
+
+    fun set(i: Int, cp: Int, fgColor: Int, bgColor: Int, attr: Int) {
+        chars[i] = cp
+        fg[i] = fgColor
+        bg[i] = bgColor
+        attrs[i] = attr
+    }
+
+    /** Blanks [from, to) using [bgColor] (background-colour-erase, like xterm). */
+    fun clear(from: Int, to: Int, bgColor: Int) {
+        val s = if (from < 0) 0 else from
+        val e = if (to > size) size else to
+        for (i in s until e) {
+            chars[i] = SPACE
+            fg[i] = 0
+            bg[i] = bgColor
+            attrs[i] = 0
+        }
+    }
+
+    fun resize(newCols: Int) {
+        val old = size
+        if (newCols == old) return
+        chars = chars.copyOf(newCols)
+        fg = fg.copyOf(newCols)
+        bg = bg.copyOf(newCols)
+        attrs = attrs.copyOf(newCols)
+        for (i in old until newCols) {
+            chars[i] = SPACE
+        }
+    }
+
+    /** ICH: shift cells right from [at], filling the gap with blanks. */
+    fun insertCells(at: Int, n: Int, bgColor: Int) {
+        val count = minOf(n, size - at)
+        if (count <= 0) return
+        for (arr in arrayOf(chars, fg, bg, attrs)) {
+            System.arraycopy(arr, at, arr, at + count, size - at - count)
+        }
+        clear(at, at + count, bgColor)
+    }
+
+    /** DCH: shift cells left over [at], blanking the tail. */
+    fun deleteCells(at: Int, n: Int, bgColor: Int) {
+        val count = minOf(n, size - at)
+        if (count <= 0) return
+        for (arr in arrayOf(chars, fg, bg, attrs)) {
+            System.arraycopy(arr, at + count, arr, at, size - at - count)
+        }
+        clear(size - count, size, bgColor)
+    }
+
+    companion object {
+        const val SPACE = 0x20
+    }

+/**
+ * A rows x cols character grid with cursor, scroll region and (optionally)
+ * scrollback. The emulator owns two of these: the normal screen (with
+ * scrollback) and the alternate screen (without).
+ *
+ * Not thread-safe: all access is expected on the UI thread.
+ */
+class TerminalBuffer(initCols: Int, initRows: Int, private val scrollbackLimit: Int) {
+
+    var cols: Int = initCols
+        private set
+    var rows: Int = initRows
+        private set
+
+    private var lines: Array<TerminalRow> = Array(initRows) { TerminalRow(initCols) }
+    private val scrollback = ArrayDeque<TerminalRow>()
+
+    var cursorX: Int = 0
+    var cursorY: Int = 0
+
+    /** Inclusive scroll region rows (DECSTBM). */
+    var scrollTop: Int = 0
+        private set
+    var scrollBottom: Int = initRows - 1
+        private set
+
+    /** True after writing in the last column: the wrap happens on the next char. */
+    var wrapPending: Boolean = false
+
+    val scrollbackSize: Int get() = scrollback.size
+
+    /**
+     * Row at [index]: 0 until rows is the live screen, negative reaches into
+     * scrollback (-1 = most recent scrolled-off line). Null when out of range.
+     * Scrollback rows may be a different width than [cols] after a resize.
+     */
+    fun getRow(index: Int): TerminalRow? {
+        if (index >= 0) {
+            return if (index < rows) lines[index] else null
+        }
+        val i = scrollback.size + index
+        return if (i >= 0) scrollback[i] else null
+    }
+
+    fun moveCursor(x: Int, y: Int) {
+        cursorX = x.coerceIn(0, cols - 1)
+        cursorY = y.coerceIn(0, rows - 1)
+        wrapPending = false
+    }
+
+    fun setScrollRegion(top: Int, bottom: Int) {
+        scrollTop = top.coerceIn(0, rows - 1)
+        scrollBottom = bottom.coerceIn(scrollTop, rows - 1)
+    }
+
+    fun resetScrollRegion() {
+        scrollTop = 0
+        scrollBottom = rows - 1
+    }
+
+    /** Writes one code point at the cursor, handling deferred wrap and wide chars. */
+    fun putChar(cp: Int, width: Int, fg: Int, bg: Int, attrs: Int, autoWrap: Boolean, insert: Boolean) {
+        var w = if (width == 2 && cols >= 2) 2 else 1
+        if (wrapPending) {
+            if (autoWrap) {
+                cursorX = 0
+                lineFeed(bg)
+            }
+            wrapPending = false
+        }
+        if (w == 2 && cursorX == cols - 1) {
+            if (autoWrap) {
+                // Not enough room for both halves: blank the last cell, wrap.
+                lines[cursorY].clear(cursorX, cursorX + 1, bg)
+                cursorX = 0
+                lineFeed(bg)
+            } else {
+                w = 1
+            }
+        }
+        val row = lines[cursorY]
+        if (insert) {
+            row.insertCells(cursorX, w, bg)
+        }
+        // Overwriting one half of an existing wide char would leave an orphan half.
+        if (cursorX > 0 && (row.attrs[cursorX] and TerminalAttr.WIDE_TAIL) != 0) {
+            row.clear(cursorX - 1, cursorX, bg)
+        }
+        if (w == 1 && cursorX + 1 < cols && (row.attrs[cursorX + 1] and TerminalAttr.WIDE_TAIL) != 0) {
+            row.clear(cursorX + 1, cursorX + 2, bg)
+        }
+        row.set(cursorX, cp, fg, bg, attrs)
+        if (w == 2) {
+            row.set(cursorX + 1, TerminalRow.SPACE, fg, bg, attrs or TerminalAttr.WIDE_TAIL)
+        }
+        cursorX += w
+        if (cursorX >= cols) {
+            cursorX = cols - 1
+            wrapPending = true
+        }
+    }
+
+    /** IND / LF: move down, scrolling the region when at its bottom. */
+    fun lineFeed(bg: Int) {
+        wrapPending = false
+        if (cursorY == scrollBottom) {
+            scrollUp(1, bg)
+        } else if (cursorY < rows - 1) {
+            cursorY++
+        }
+    }
+
+    /** RI: move up, scrolling the region down when at its top. */
+    fun reverseLineFeed(bg: Int) {
+        wrapPending = false
+        if (cursorY == scrollTop) {
+            scrollDown(1, bg)
+        } else if (cursorY > 0) {
+            cursorY--
+        }
+    }
+
+    fun scrollUp(n: Int, bg: Int) {
+        val count = minOf(n, scrollBottom - scrollTop + 1)
+        // Only whole-screen scrolling feeds scrollback; a TUI's inner region
+        // (e.g. a status bar split) shouldn't pollute it.
+        val toScrollback = scrollbackLimit > 0 && scrollTop == 0 && scrollBottom == rows - 1
+        repeat(count) {
+            val top = lines[scrollTop]
+            for (i in scrollTop until scrollBottom) {
+                lines[i] = lines[i + 1]
+            }
+            if (toScrollback) {
+                scrollback.addLast(top)
+                if (scrollback.size > scrollbackLimit) {
+                    scrollback.removeFirst()
+                }
+                val fresh = TerminalRow(cols)
+                fresh.clear(0, cols, bg)
+                lines[scrollBottom] = fresh
+            } else {
+                top.clear(0, cols, bg)
+                lines[scrollBottom] = top
+            }
+        }
+    }
+
+    fun scrollDown(n: Int, bg: Int) {
+        val count = minOf(n, scrollBottom - scrollTop + 1)
+        repeat(count) {
+            val bottom = lines[scrollBottom]
+            for (i in scrollBottom downTo scrollTop + 1) {
+                lines[i] = lines[i - 1]
+            }
+            bottom.clear(0, cols, bg)
+            lines[scrollTop] = bottom
+        }
+    }
+
+    /** IL */
+    fun insertLines(n: Int, bg: Int) {
+        if (cursorY < scrollTop || cursorY > scrollBottom) return
+        val count = minOf(n, scrollBottom - cursorY + 1)
+        repeat(count) {
+            val bottom = lines[scrollBottom]
+            for (i in scrollBottom downTo cursorY + 1) {
+                lines[i] = lines[i - 1]
+            }
+            bottom.clear(0, cols, bg)
+            lines[cursorY] = bottom
+        }
+        cursorX = 0
+        wrapPending = false
+    }
+
+    /** DL */
+    fun deleteLines(n: Int, bg: Int) {
+        if (cursorY < scrollTop || cursorY > scrollBottom) return
+        val count = minOf(n, scrollBottom - cursorY + 1)
+        repeat(count) {
+            val removed = lines[cursorY]
+            for (i in cursorY until scrollBottom) {
+                lines[i] = lines[i + 1]
+            }
+            removed.clear(0, cols, bg)
+            lines[scrollBottom] = removed
+        }
+        cursorX = 0
+        wrapPending = false
+    }
+
+    fun insertChars(n: Int, bg: Int) {
+        lines[cursorY].insertCells(cursorX, n, bg)
+        wrapPending = false
+    }
+
+    fun deleteChars(n: Int, bg: Int) {
+        lines[cursorY].deleteCells(cursorX, n, bg)
+        wrapPending = false
+    }
+
+    fun eraseChars(n: Int, bg: Int) {
+        lines[cursorY].clear(cursorX, minOf(cols, cursorX + n), bg)
+    }
+
+    /** EL: 0 = cursor to end, 1 = start to cursor, 2 = whole line. */
+    fun eraseInLine(mode: Int, bg: Int) {
+        val row = lines[cursorY]
+        when (mode) {
+            0 -> row.clear(cursorX, cols, bg)
+            1 -> row.clear(0, cursorX + 1, bg)
+            2 -> row.clear(0, cols, bg)
+        }
+    }
+
+    /** ED: 0 = cursor to end, 1 = start to cursor, 2 = all, 3 = all + scrollback. */
+    fun eraseInDisplay(mode: Int, bg: Int) {
+        when (mode) {
+            0 -> {
+                eraseInLine(0, bg)
+                for (y in cursorY + 1 until rows) lines[y].clear(0, cols, bg)
+            }
+            1 -> {
+                for (y in 0 until cursorY) lines[y].clear(0, cols, bg)
+                eraseInLine(1, bg)
+            }
+            2 -> clearAll(bg)
+            3 -> {
+                clearAll(bg)
+                scrollback.clear()
+            }
+        }
+    }
+
+    fun clearAll(bg: Int) {
+        for (row in lines) row.clear(0, cols, bg)
+    }
+
+    /** RIS for this buffer. Scrollback is intentionally kept. */
+    fun fullReset() {
+        lines = Array(rows) { TerminalRow(cols) }
+        cursorX = 0
+        cursorY = 0
+        wrapPending = false
+        resetScrollRegion()
+    }
+
+    /**
+     * Resizes the grid without reflow (the remote app redraws on SIGWINCH).
+     * Shrinking pushes the top lines into scrollback so the cursor line stays
+     * visible; growing pulls them back, so showing/hiding the soft keyboard
+     * round-trips cleanly.
+     */
+    fun resize(newCols: Int, newRows: Int) {
+        if (newCols == cols && newRows == rows) return
+        val list = lines.toMutableList()
+        var cy = cursorY
+        if (newRows < rows) {
+            // Drop lines below the cursor first, then scroll the top away.
+            val removeBottom = minOf(rows - newRows, rows - 1 - cursorY)
+            repeat(removeBottom) { list.removeAt(list.size - 1) }
+            val pushTop = rows - newRows - removeBottom
+            repeat(pushTop) {
+                val r = list.removeAt(0)
+                if (scrollbackLimit > 0) {
+                    scrollback.addLast(r)
+                    if (scrollback.size > scrollbackLimit) {
+                        scrollback.removeFirst()
+                    }
+                }
+                cy--
+            }
+        } else if (newRows > rows) {
+            val grow = newRows - rows
+            val pull = minOf(grow, scrollback.size)
+            repeat(pull) {
+                list.add(0, scrollback.removeLast())
+                cy++
+            }
+            repeat(grow - pull) { list.add(TerminalRow(cols)) }
+        }
+        for (r in list) r.resize(newCols)
+        lines = list.toTypedArray()
+        cols = newCols
+        rows = newRows
+        cursorY = cy.coerceIn(0, rows - 1)
+        cursorX = cursorX.coerceIn(0, cols - 1)
+        wrapPending = false
+        resetScrollRegion()
+    }
+}

+package jp.hi7.sshtermproto.terminal

+/**
+ * Colour handling for the terminal grid.
+ *
+ * Each cell stores its colour as a single Int so a row needs no per-cell
+ * objects:
+ *   0                  -> "default" (resolved to the theme fg/bg at draw time)
+ *   0x01_000000 | idx  -> palette index 0..255
+ *   0xFF_RRGGBB        -> 24-bit colour (already ARGB, usable as-is)
+ */
+object TerminalColors {
+    const val DEFAULT = 0
+    private const val PALETTE_TAG = 0x01
+
+    private fun opaque(rgb: Int): Int = (0xFF shl 24) or rgb
+
+    // Default text is light grey (not the old all-green log look): TUIs such
+    // as Claude Code / Codex rely on the default fg being distinguishable
+    // from the ANSI green they use for highlights.
+    val DEFAULT_FG: Int = opaque(0xE5E5E5)
+    val DEFAULT_BG: Int = opaque(0x000000)
+
+    // ANSI 0..7 (normal) followed by 8..15 (bright).
+    private val ANSI16 = intArrayOf(
+        0x000000, 0xCD3131, 0x0DBC79, 0xE5E510,
+        0x2472C8, 0xBC3FBC, 0x11A8CD, 0xE5E5E5,
+        0x666666, 0xF14C4C, 0x23D18B, 0xF5F543,
+        0x3B8EEA, 0xD670D6, 0x29B8DB, 0xFFFFFF
+    )
+
+    private val palette256: IntArray = buildPalette()
+
+    private fun buildPalette(): IntArray {
+        val p = IntArray(256)
+        for (i in 0 until 16) {
+            p[i] = opaque(ANSI16[i])
+        }
+        // 6x6x6 colour cube (16..231)
+        val levels = intArrayOf(0, 95, 135, 175, 215, 255)
+        for (i in 0 until 216) {
+            val r = levels[i / 36]
+            val g = levels[(i / 6) % 6]
+            val b = levels[i % 6]
+            p[16 + i] = opaque((r shl 16) or (g shl 8) or b)
+        }
+        // 24-step grey ramp (232..255)
+        for (i in 0 until 24) {
+            val v = 8 + i * 10
+            p[232 + i] = opaque((v shl 16) or (v shl 8) or v)
+        }
+        return p
+    }
+
+    fun palette(index: Int): Int = (PALETTE_TAG shl 24) or (index and 0xFF)
+
+    fun rgb(r: Int, g: Int, b: Int): Int =
+        opaque(((r and 0xFF) shl 16) or ((g and 0xFF) shl 8) or (b and 0xFF))
+
+    /** Turns a stored cell colour into an ARGB value for Paint. */
+    fun resolve(color: Int, default: Int): Int {
+        if (color == DEFAULT) return default
+        if ((color ushr 24) == PALETTE_TAG) return palette256[color and 0xFF]
+        return color
+    }
+}

+/**
+ * A small, from-scratch VT100 / xterm-subset parser that drives two
+ * [TerminalBuffer]s (normal + alternate screen).
+ *
+ * Written independently (no GPL terminal-emulator code) so the repository
+ * can stay Apache-2.0.
+ *
+ * Input is already-decoded text (the caller decodes UTF-8), fed through
+ * [feed]. Anything unrecognised is skipped without throwing; a bad sequence
+ * can never wedge the parser because ESC / CAN / SUB always resync it.
+ *
+ * Replies the remote may be waiting for (cursor position report, device
+ * attributes) are delivered through [responder].
+ *
+ * Not thread-safe: call everything from the UI thread.
+ */
+class TerminalEmulator(initialCols: Int, initialRows: Int) {

+        private const val SCROLLBACK_LINES = 2000

+        private const val MAX_PARAM_VALUE = 99999

+    }

+    private var insertMode = false

+    private var curFg = TerminalColors.DEFAULT
+    private var curBg = TerminalColors.DEFAULT
+    private var curAttrs = 0
+
+    // --- character sets (only DEC line drawing vs. ASCII) ---
+    private var g0Graphics = false
+    private var g1Graphics = false
+    private var shiftOut = false

+    private class SavedCursor(
+        val x: Int,
+        val y: Int,

+        val attrs: Int,
+        val g0: Boolean,

+
+    private var state = State.GROUND

+    private val colonAfter = BooleanArray(MAX_PARAMS)
+    private var paramCount = 0
+    private var curParam = -1
+    private var csiPrefix = 0
+    private var csiIntermediate = 0
+    private var escIntermediate = 0
+    private var pendingHigh = 0 // high surrogate waiting for its low half

+    // ------------------------------------------------------------------
+    // Public API
+    // ------------------------------------------------------------------

+    fun feed(text: CharSequence) {
+        for (i in 0 until text.length) {
+            val ch = text[i]
+            try {
+                if (pendingHigh != 0) {
+                    if (Character.isLowSurrogate(ch)) {
+                        val cp = Character.toCodePoint(pendingHigh.toChar(), ch)
+                        pendingHigh = 0
+                        process(cp)
+                        continue
+                    }
+                    pendingHigh = 0 // lone high surrogate: drop it, handle ch normally
+                }
+                if (Character.isHighSurrogate(ch)) {
+                    pendingHigh = ch.code
+                    continue
+                }
+                if (Character.isLowSurrogate(ch)) continue // lone low surrogate
+                process(ch.code)
+            } catch (e: Exception) {
+                // Never let a malformed sequence escape; resync to ground.
+                state = State.GROUND
+            }
+        }
+    }

+    fun resize(cols: Int, rows: Int) {
+        mainBuffer.resize(cols, rows)
+        altBuffer.resize(cols, rows)
+    }

+    /** RIS: full reset (scrollback of the normal screen is kept). */

+        pendingHigh = 0
+        buffer = mainBuffer
+        mainBuffer.fullReset()
+        altBuffer.fullReset()
+        resetAttrs()

+        g1Graphics = false

+        cursorVisible = true

+        bracketedPaste = false

+    // Parser core

+    private fun process(cp: Int) {
+        // ESC and CAN/SUB act in every state, so a corrupt sequence can't stick.
+        if (cp == 0x1B) {
+            state = State.ESCAPE // also terminates OSC / string (ESC \ = ST)
+            escIntermediate = 0
+            return
+        }
+        if (cp == 0x18 || cp == 0x1A) {
+            state = State.GROUND
+            return
+        }
+        if (state == State.OSC || state == State.STRING) {
+            if (cp == 0x07) state = State.GROUND // BEL terminates OSC
+            return // OSC / DCS / APC payloads are ignored
+        }
+        if (cp < 0x20) {
+            control(cp) // C0 controls execute even in the middle of a CSI
+            return
+        }
+        when (state) {
+            State.GROUND -> ground(cp)
+            State.ESCAPE -> escape(cp)
+            State.ESC_INTERMEDIATE -> escIntermediateState(cp)
+            State.CSI -> csi(cp)
+            else -> state = State.GROUND
+        }
+    }

+    private fun control(cp: Int) {
+        val b = buffer
+        when (cp) {
+            0x08 -> { // BS
+                b.cursorX = maxOf(0, b.cursorX - 1)
+                b.wrapPending = false
+            }

+            0x0A, 0x0B, 0x0C -> b.lineFeed(curBg) // LF / VT / FF
+            0x0D -> { // CR
+                b.cursorX = 0
+                b.wrapPending = false
+            }
+            0x0E -> shiftOut = true  // SO
+            0x0F -> shiftOut = false // SI
+            else -> { /* BEL and the rest: ignored on purpose */ }
+        }
+    }

+    private fun ground(cp: Int) {
+        if (cp == 0x7F || cp in 0x80..0x9F) return
+        var c = cp
+        val graphics = if (shiftOut) g1Graphics else g0Graphics
+        if (graphics && c in 0x60..0x7E) {
+            c = DEC_GRAPHICS[c - 0x60].code
+        }
+        val w = CharWidth.of(c)
+        if (w == 0) return // combining marks are dropped (no grapheme clusters)
+        buffer.putChar(c, w, curFg, curBg, curAttrs, autoWrap, insertMode)
+    }

+    private fun escape(cp: Int) {
+        when (cp) {
+            '['.code -> {
+                state = State.CSI
+                paramCount = 0
+                curParam = -1
+                csiPrefix = 0
+                csiIntermediate = 0
+            }
+            ']'.code -> state = State.OSC
+            'P'.code, 'X'.code, '^'.code, '_'.code -> state = State.STRING
+            in 0x20..0x2F -> {
+                escIntermediate = cp
+                state = State.ESC_INTERMEDIATE
+            }
+            else -> {
+                state = State.GROUND
+                escDispatch(cp)
+            }
+        }
+    }

+            escIntermediate = cp
+            return
+        }
+        state = State.GROUND
+        when (escIntermediate) {
+            '('.code -> g0Graphics = (cp == '0'.code)
+            ')'.code -> g1Graphics = (cp == '0'.code)
+            else -> { /* #8 (DECALN), %G etc.: ignored */ }
+        }
+    }
+
+    private fun escDispatch(cp: Int) {
+        val b = buffer
+        when (cp) {
+            '7'.code -> savedCursor = captureCursor()
+            '8'.code -> restoreSavedCursor()
+            'D'.code -> b.lineFeed(curBg)
+            'E'.code -> {

+            }
+            'M'.code -> b.reverseLineFeed(curBg)
+            'c'.code -> reset()
+            else -> { /* '=', '>', 'H', '\\' ...: ignored */ }
+        }
+    }

+    private fun csi(cp: Int) {

+                curParam = minOf(base * 10 + (cp - 0x30), MAX_PARAM_VALUE)
+            }
+            cp == ';'.code -> pushParam(false)
+            cp == ':'.code -> pushParam(true)
+            cp in 0x3C..0x3F -> { // < = > ?  private-mode prefix
+                if (paramCount == 0 && curParam < 0 && csiPrefix == 0) {
+                    csiPrefix = cp
+                }
+            }
+            cp in 0x20..0x2F -> csiIntermediate = cp
+            cp in 0x40..0x7E -> {
+                state = State.GROUND
+                if (curParam >= 0 || paramCount > 0) pushParam(false)
+                dispatchCsi(cp)
+            }
+            else -> state = State.GROUND // 0x7F / non-ASCII inside CSI: abort
+        }
+    }
+
+    private fun pushParam(colon: Boolean) {
+        if (paramCount < MAX_PARAMS) {
+            params[paramCount] = curParam
+            colonAfter[paramCount] = colon
+            paramCount++
+        }
+        curParam = -1
+    }

+    // ------------------------------------------------------------------

+            'D'.code -> b.moveCursor(b.cursorX - p(0, 1), b.cursorY)

+            'G'.code, '`'.code -> b.moveCursor(p(0, 1) - 1, b.cursorY)
+            'd'.code -> b.moveCursor(b.cursorX, p(0, 1) - 1)
+            'H'.code, 'f'.code -> b.moveCursor(p(1, 1) - 1, p(0, 1) - 1)
+            'I'.code -> repeat(minOf(p(0, 1), b.cols)) { tabForward() }
+            'Z'.code -> repeat(minOf(p(0, 1), b.cols)) { tabBackward() }
+            'J'.code -> b.eraseInDisplay(p(0, 0), curBg)
+            'K'.code -> b.eraseInLine(p(0, 0), curBg)
+            'L'.code -> b.insertLines(p(0, 1), curBg)
+            'M'.code -> b.deleteLines(p(0, 1), curBg)
+            'P'.code -> b.deleteChars(p(0, 1), curBg)
+            '@'.code -> b.insertChars(p(0, 1), curBg)
+            'X'.code -> b.eraseChars(p(0, 1), curBg)
+            'S'.code -> b.scrollUp(p(0, 1), curBg)
+            'T'.code -> {

+            }
+            'm'.code -> sgr()
+            'r'.code -> setScrollRegion()
+            's'.code -> {
+                if (paramCount == 0) savedCursor = captureCursor()
+            }
+            'u'.code -> restoreSavedCursor()
+            'n'.code -> deviceStatusReport()
+            'c'.code -> {
+                if (p(0, 0) == 0) respond("\u001b[?1;2c") // DA1: "VT100 with AVO"
+            }
+            'h'.code -> {
+                if (p(0, 0) == 4) insertMode = true
+            }
+            'l'.code -> {
+                if (p(0, 0) == 4) insertMode = false
+            }
+            else -> { /* unknown: ignored */ }
+        }
+    }
+
+    private fun dispatchPrivateCsi(fin: Int) {
+        when (csiPrefix) {
+            '?'.code -> when (fin) {
+                'h'.code -> for (i in 0 until paramCount) setPrivateMode(params[i], true)
+                'l'.code -> for (i in 0 until paramCount) setPrivateMode(params[i], false)
+                else -> { /* ?u (kitty kbd), ?$p (DECRQM) ...: ignored */ }
+            }
+            '>'.code -> {

+                // purpose: they must NOT be mistaken for SGR.
+                if (fin == 'c'.code) respond("\u001b[>0;0;0c") // DA2
+            }
+            else -> { /* '<' and '=' prefixes: ignored */ }
+        }
+    }
+
+    private fun setPrivateMode(mode: Int, on: Boolean) {
+        when (mode) {
+            1 -> appCursorKeys = on
+            7 -> autoWrap = on
+            25 -> cursorVisible = on
+            47, 1047 -> switchScreen(on, false)
+            1049 -> switchScreen(on, true)
+            2004 -> bracketedPaste = on
+            else -> { /* mouse modes, focus events, sync output ...: ignored */ }
+        }
+    }
+
+    private fun switchScreen(alt: Boolean, saveRestoreCursor: Boolean) {
+        if (alt) {
+            if (buffer === altBuffer) return
+            if (saveRestoreCursor) savedCursor1049 = captureCursor()

+            altBuffer.moveCursor(mainBuffer.cursorX, mainBuffer.cursorY)
+            if (saveRestoreCursor) altBuffer.clearAll(TerminalColors.DEFAULT)
+            buffer = altBuffer
+        } else {
+            if (buffer !== altBuffer) return
+            buffer = mainBuffer
+            if (saveRestoreCursor) {
+                savedCursor1049?.let { applyCursor(it) }
+            }
+        }
+    }

+    private fun cursorUp(n: Int) {
+        val b = buffer
+        val top = if (b.cursorY >= b.scrollTop) b.scrollTop else 0
+        b.moveCursor(b.cursorX, maxOf(top, b.cursorY - n))
+    }
+
+    private fun cursorDown(n: Int) {
+        val b = buffer
+        val bottom = if (b.cursorY <= b.scrollBottom) b.scrollBottom else b.rows - 1
+        b.moveCursor(b.cursorX, minOf(bottom, b.cursorY + n))
+    }
+
+    private fun tabForward() {
+        val b = buffer
+        var x = b.cursorX + 1
+        while (x % 8 != 0) x++
+        b.moveCursor(minOf(x, b.cols - 1), b.cursorY)
+    }
+
+    private fun tabBackward() {
+        val b = buffer
+        var x = b.cursorX - 1
+        while (x > 0 && x % 8 != 0) x--
+        b.moveCursor(maxOf(0, x), b.cursorY)
+    }
+
+    private fun setScrollRegion() {
+        val b = buffer
+        val top = p(0, 1) - 1
+        val bottom = minOf(p(1, b.rows), b.rows) - 1
+        if (top < bottom) {
+            b.setScrollRegion(top, bottom)
+            b.moveCursor(0, 0)
+        }
+    }
+
+    private fun deviceStatusReport() {
+        when (p(0, 0)) {
+            5 -> respond("\u001b[0n")
+            6 -> respond("\u001b[${buffer.cursorY + 1};${buffer.cursorX + 1}R")
+        }
+    }
+
+    private fun respond(s: String) {
+        responder?.invoke(s)
+    }
+
+    // ------------------------------------------------------------------
+    // Cursor save / restore, soft reset
+    // ------------------------------------------------------------------
+
+    private fun captureCursor(): SavedCursor =
+        SavedCursor(buffer.cursorX, buffer.cursorY, curFg, curBg, curAttrs, g0Graphics, g1Graphics, shiftOut)
+
+    private fun applyCursor(s: SavedCursor) {
+        buffer.moveCursor(s.x, s.y)
+        curFg = s.fg
+        curBg = s.bg
+        curAttrs = s.attrs
+        g0Graphics = s.g0
+        g1Graphics = s.g1
+        shiftOut = s.shiftOut
+    }
+
+    private fun restoreSavedCursor() {
+        val s = savedCursor
+        if (s != null) {
+            applyCursor(s)
+        } else {
+            buffer.moveCursor(0, 0)
+        }
+    }
+
+    private fun softReset() {
+        resetAttrs()
+        cursorVisible = true
+        appCursorKeys = false
+        autoWrap = true
+        insertMode = false
+        g0Graphics = false
+        g1Graphics = false
+        shiftOut = false
+        savedCursor = null
+        buffer.resetScrollRegion()
+    }
+
+    // ------------------------------------------------------------------
+    // SGR
+    // ------------------------------------------------------------------
+
+    private fun resetAttrs() {
+        curFg = TerminalColors.DEFAULT
+        curBg = TerminalColors.DEFAULT
+        curAttrs = 0
+    }
+
+    private fun setColor(foreground: Boolean, color: Int) {
+        if (foreground) curFg = color else curBg = color
+    }
+
+    private fun sgr() {
+        if (paramCount == 0) {
+            resetAttrs()
+            return
+        }
+        var i = 0
+        while (i < paramCount) {
+            // A run of params joined by ':' is one sub-parameter group (i..j).
+            var j = i
+            while (j < paramCount - 1 && colonAfter[j]) j++
+            if (j > i) {
+                sgrColonGroup(i, j)
+                i = j + 1
+                continue
+            }
+
+            val code = if (params[i] < 0) 0 else params[i]
+            when (code) {
+                0 -> resetAttrs()
+                1 -> curAttrs = curAttrs or TerminalAttr.BOLD
+                2 -> curAttrs = curAttrs or TerminalAttr.DIM
+                3 -> curAttrs = curAttrs or TerminalAttr.ITALIC
+                4, 21 -> curAttrs = curAttrs or TerminalAttr.UNDERLINE
+                7 -> curAttrs = curAttrs or TerminalAttr.INVERSE
+                8 -> curAttrs = curAttrs or TerminalAttr.INVISIBLE
+                9 -> curAttrs = curAttrs or TerminalAttr.STRIKE
+                22 -> curAttrs = curAttrs and (TerminalAttr.BOLD or TerminalAttr.DIM).inv()
+                23 -> curAttrs = curAttrs and TerminalAttr.ITALIC.inv()
+                24 -> curAttrs = curAttrs and TerminalAttr.UNDERLINE.inv()
+                27 -> curAttrs = curAttrs and TerminalAttr.INVERSE.inv()
+                28 -> curAttrs = curAttrs and TerminalAttr.INVISIBLE.inv()
+                29 -> curAttrs = curAttrs and TerminalAttr.STRIKE.inv()
+                in 30..37 -> curFg = TerminalColors.palette(code - 30)
+                39 -> curFg = TerminalColors.DEFAULT
+                in 40..47 -> curBg = TerminalColors.palette(code - 40)
+                49 -> curBg = TerminalColors.DEFAULT
+                in 90..97 -> curFg = TerminalColors.palette(code - 90 + 8)
+                in 100..107 -> curBg = TerminalColors.palette(code - 100 + 8)
+                38, 48 -> {
+                    // Semicolon form: 38;5;n  or  38;2;r;g;b
+                    val mode = if (i + 1 < paramCount) params[i + 1] else -1
+                    if (mode == 5 && i + 2 < paramCount) {
+                        setColor(code == 38, TerminalColors.palette(params[i + 2].coerceIn(0, 255)))
+                        i += 2
+                    } else if (mode == 2 && i + 4 < paramCount) {
+                        setColor(
+                            code == 38,
+                            TerminalColors.rgb(
+                                params[i + 2].coerceIn(0, 255),
+                                params[i + 3].coerceIn(0, 255),
+                                params[i + 4].coerceIn(0, 255)
+                            )
+                        )
+                        i += 4
+                    } else {
+                        i = paramCount // malformed: swallow the rest
+                    }
+                }
+                else -> { /* blink (5/6), fonts, ...: ignored */ }
+            }
+            i++
+        }
+    }
+
+    /** Colon sub-parameters: 38:5:n, 38:2:[cs]:r:g:b, 4:n (underline style). */
+    private fun sgrColonGroup(i: Int, j: Int) {
+        val code = params[i]
+        when (code) {
+            38, 48 -> {
+                val mode = params[i + 1]
+                if (mode == 5 && j >= i + 2) {
+                    setColor(code == 38, TerminalColors.palette(params[i + 2].coerceIn(0, 255)))
+                } else if (mode == 2) {
+                    // 38:2:r:g:b (4 values after) or 38:2:<colourspace>:r:g:b (5 values)
+                    val start = if (j - i >= 5) i + 3 else i + 2
+                    if (j >= start + 2) {
+                        setColor(
+                            code == 38,
+                            TerminalColors.rgb(
+                                params[start].coerceIn(0, 255),
+                                params[start + 1].coerceIn(0, 255),
+                                params[start + 2].coerceIn(0, 255)
+                            )
+                        )
+                    }
+                }
+            }
+            4 -> {
+                curAttrs = if (params[i + 1] == 0) {
+                    curAttrs and TerminalAttr.UNDERLINE.inv()
+                } else {
+                    curAttrs or TerminalAttr.UNDERLINE
+                }
+            }
+            else -> { /* ignored */ }
+        }
+    }
+}
